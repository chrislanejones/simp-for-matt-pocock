Page memory with sessionStorage

BoxHarbor's src/lib/session-cache.ts and how pages use it.

The recipe for a new page

const savedKey = `mypage:${id}`;          // everything that identifies the data
const [restoredKey, setRestoredKey] = useState<string | null>(null);

useEffect(() => {                           // restore
  const t = setTimeout(() => {
    const saved = readSession<Saved>(savedKey, 6 * HOUR)?.value;
    if (saved) { /* set each piece of state */ }
    setRestoredKey(savedKey);
  }, 0);
  return () => clearTimeout(t);
}, [savedKey]);

useEffect(() => {                           // save, gated
  if (restoredKey === savedKey) writeSession(savedKey, { /* state */ });
}, [restoredKey, savedKey /* , each piece of state */]);
